# 모든 리그 전체 수집
python poe2_overquality-flip.py all --league all

# 모든 리그, 마지막 수집 후 6시간 지난 리그만
python poe2_overquality-flip.py all --league all:6

# 모든 리그 + 수집 중지된 항목까지 전부 다시 조회
python poe2_overquality-flip.py all --league all --retry-stopped

# 환율만 빠르게 갱신 (GGG 검색 없음)
python poe2_overquality-flip.py --rates-only --league all

--retry-stopped	all 실행에서도 수집 중지 항목을 다시 조회합니다.
--league latest	기본값입니다. 거래소 API가 주는 리그 중 Standard·하드코어·SSF·Ruthless를 뺀 첫 번째 리그를 씁니다.
--league "리그명"	리그 이름을 직접 지정합니다. 쉼표로 여러 개를 줄 수 있습니다(--league "A,B"). standard와 previous는 막혀 있습니다(BLOCKED_CHOICES).
--out 경로	저장 위치를 지정합니다. 기본은 스크립트 옆 out/prices.json이고, 파일이 있으면 병합합니다.
--no-rates	poe.ninja 환율 갱신을 건너뜁니다. 이상치 판정에는 prices.json에 저장된 이전 환율을 씁니다.
--rates-only	환율만 갱신하고 끝냅니다. GGG 검색이 없어 몇 초면 끝납니다.N=not implemented (do not read these data)
